\begin{frame}{Solución I: Circuitos en DC}
    \textbf{1. Equivalente de Thévenin:}
    $$V_{th} = V_S \cdot \left( \frac{R_3}{R_1 + R_2 + R_3} \right) = 12 \cdot \left( \frac{3.3}{6.5} \right) = 6.0\text{ V}$$
    $$R_{th} = R_3 \parallel (R_1 + R_2) = 3.3\text{ k}\Omega \parallel 3.2\text{ k}\Omega = 1.625\text{ k}\Omega$$

    $$V_L = V_{th} \cdot \left( \frac{R_L}{R_{th} + R_L} \right) = 6.0 \cdot \left( \frac{1}{2.625} \right) = 2.285\text{ V}$$

\begin{frame}{Solución III: BJT en DC}
    \textbf{Punto de Operación Q (Asumiendo $\beta$ alta):}
    $$V_B \approx V_{CC} \cdot \left( \frac{R_2}{R_1 + R_2} \right) = 15 \cdot \left( \frac{10}{43} \right) = 3.488\text{ V}$$
    $$V_E = V_B - 0.7\text{ V} = 2.788\text{ V}$$
    $$I_{CQ} \approx I_E = \frac{V_E}{R_E} = \frac{2.788\text{ V}}{1\text{ k}\Omega} = 2.788\text{ mA}$$
    $$V_C = V_{CC} - I_{CQ} \cdot R_C = 15 - (2.788\text{ mA} \cdot 2.2\text{ k}\Omega) = 8.866\text{ V}$$
    $$V_{CEQ} = V_C - V_E = 8.866\text{ V} - 2.788\text{ V} = 6.078\text{ V}$$
    \vspace{0.2cm}
    \textbf{Región:} Activa Lineal ($V_{CEQ} = 6.078\text{ V} > 0.3\text{ V}$ y la unión Base-Emisor en directa).
\end{frame}

\begin{frame}{Solución IV: Respuesta Frecuencial}
    \textbf{Función de Transferencia y Frecuencia de Corte:}
    $$H(s) = \frac{1}{1 + s R C} \quad \implies \quad f_c = \frac{1}{2\pi R C} = 159.15\text{ Hz}$$
    \vspace{0.1cm}
    
    \textbf{Cálculo analítico del desfase en $f_c$:}
    Sustituyendo $s = j\omega$ para régimen senoidal permanente, y sabiendo que en la frecuencia de corte $\omega_c = \frac{1}{RC}$:
    $$H(j\omega_c) = \frac{1}{1 + j\omega_c R C} = \frac{1}{1 + j\left(\frac{1}{RC}\right) R C} = \frac{1}{1 + j}$$
    El desfase $\phi$ corresponde al ángulo de la función de transferencia compleja:
    $$\phi = \angle H(j\omega_c) = \angle(1) - \angle(1 + j)$$
    $$\phi = 0^\circ - \arctan\left(\frac{\text{Im}}{\text{Re}}\right) = 0^\circ - \arctan(1) = -45^\circ$$
\end{frame}

\begin{frame}{Solución V: Computación Científica}
    \textbf{Análisis del Algoritmo en Python:}
    \begin{itemize}
        \item \textbf{Emulación Física:} El bloque de código \texttt{if v\_in[i] > 0.7} modela numéricamente la curva de conducción idealizada por tramos de un \textbf{diodo de silicio} en una topología de \textbf{rectificador de media onda}.
        \item \textbf{Anulación de semiciclo:} La condición \texttt{else: v\_out[i] = 0.0} bloquea el paso de corriente, emulando la polarización inversa del semiconductor.
