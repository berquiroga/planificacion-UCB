\section*{2. Plan de Medición Estricto (NO EJECUTAR SIN FIRMA)[cite: 30]}
\begin{enumerate}
    \setlength\itemsep{0.2em}
    \item Revisar driver BJT con alimentación limitada y \textbf{sin MOSFET/carga}[cite: 30]. Comprobar niveles de salida y consumo estático[cite: 30].
    \item Conectar transistor de potencia pero mantener el \textbf{BUS APAGADO} ($V_{bus} = 0\,\text{V}$). Medir $V_{GS}$ y asegurar que apaga a cero[cite: 30].
    \item \textbf{Solo si G2 está firmado:} Energizar bus principal. Registrar $V_{GS}, V_{DS}, I_D$ y transitorios. Mantener las mismas escalas que en E1[cite: 30].
\end{enumerate}
